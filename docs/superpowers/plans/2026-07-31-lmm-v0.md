```python
"""Yaşayan Bellek: LMM'in bilgi organı. Bilgi ağırlıklarda değil burada yaşar."""
import time


class DonguHatasi(Exception):
    """Tür hiyerarşisinde döngü oluşturacak yazım."""


class Kenar:
    def __init__(self, kavram, iliski, hedef, kaynak="bilinmiyor",
                 guven=0.6, istisna=False, zaman=None):
        self.kavram = kavram
        self.iliski = iliski        # "tür" | "yapabilir" | "yapamaz"
        self.hedef = hedef
        self.kaynak = kaynak
        self.guven = guven
        self.istisna = istisna
        self.zaman = zaman if zaman is not None else time.time()

    def sozluk(self):
        return {"kavram": self.kavram, "iliski": self.iliski, "hedef": self.hedef,
                "kaynak": self.kaynak, "guven": self.guven,
                "istisna": self.istisna, "zaman": self.zaman}

    @staticmethod
    def sozlukten(d):
        return Kenar(**d)


class Bellek:
    def __init__(self):
        self.kenarlar = []

    def sorgula(self, kavram, iliski=None):
        return [k for k in self.kenarlar
                if k.kavram == kavram and (iliski is None or k.iliski == iliski)]

    def dogrudan(self, kavram, iliski, hedef):
        for k in self.kenarlar:
            if k.kavram == kavram and k.iliski == iliski and k.hedef == hedef:
                return k
        return None

    def yaz(self, kenar):
        if kenar.iliski == "tür" and self._dongu_olusur(kenar):
            raise DonguHatasi(f"{kenar.kavram} -> {kenar.hedef} döngü oluşturur")
        mevcut = self.dogrudan(kenar.kavram, kenar.iliski, kenar.hedef)
        if mevcut is not None:
            mevcut.guven = min(1.0, mevcut.guven + 0.2)
            return mevcut
        self.kenarlar.append(kenar)
        return kenar

    def _dongu_olusur(self, kenar):
        # kenar.hedef'ten tür kenarlarını izleyerek kenar.kavram'a ulaşılıyor mu?
        kuyruk, gorulen = [kenar.hedef], set()
        while kuyruk:
            su = kuyruk.pop(0)
            if su == kenar.kavram:
                return True
            if su in gorulen:
                continue
            gorulen.add(su)
            kuyruk.extend(k.hedef for k in self.sorgula(su, "tür"))
        return False
```

        gecici = yol + ".tmp"
        with open(gecici, "w", encoding="utf-8") as f:

"""Muhakeme Motoru: bellek üstünde kalıtım, istisna ve çelişki çıkarımı."""


class Muhakeme:
    def __init__(self, bellek):
        self.bellek = bellek

    def atalar(self, kavram):
        """Tür kenarlarını izleyerek ataları yakından uzağa sırayla döndürür."""
        sonuc, kuyruk, gorulen = [], [kavram], {kavram}
        while kuyruk:
            su = kuyruk.pop(0)
            for k in self.bellek.sorgula(su, "tür"):
                if k.hedef not in gorulen:
                    gorulen.add(k.hedef)
                    sonuc.append(k.hedef)
                    kuyruk.append(k.hedef)
        return sonuc

    def yapabilir_mi(self, kavram, eylem):
        """(True|False|None, açıklama zinciri). None = bellekte hiç bilgi yok."""
        d = self.bellek.dogrudan(kavram, "yapamaz", eylem)
        if d:
            return False, [f"{kavram} {eylem} yapamaz (doğrudan bilgi, kaynak: {d.kaynak})"]
        d = self.bellek.dogrudan(kavram, "yapabilir", eylem)
        if d:
            return True, [f"{kavram} {eylem} yapabilir (doğrudan bilgi, kaynak: {d.kaynak})"]
        for ata in self.atalar(kavram):      # yakın ata uzak atayı ezer
            d = self.bellek.dogrudan(ata, "yapamaz", eylem)
            if d:
                return False, [f"{kavram} bir {ata}", f"{ata} {eylem} yapamaz"]
            d = self.bellek.dogrudan(ata, "yapabilir", eylem)
            if d:
                return True, [f"{kavram} bir {ata}", f"{ata} {eylem} yapabilir"]
        return None, []
```

"""Dil Sezgisi Çekirdeği: cümle -> Niyet. Bilgi taşımaz, yalnızca dil."""

NOKTALAMA = ".,!?;:\"'"

# v0 kontrollü dünyanın fiil sözlüğü: yüzey biçim -> (mastar, yapabilir_mi)
FIIL_SOZLUGU = {
    "uçar": ("uçmak", True), "uçamaz": ("uçmak", False),
    "yüzer": ("yüzmek", True), "yüzemez": ("yüzmek", False),
    "koşar": ("koşmak", True), "koşamaz": ("koşmak", False),
    "okur": ("okumak", True), "okuyamaz": ("okumak", False),
    "içer": ("içmek", True), "içemez": ("içmek", False),
    "konuşur": ("konuşmak", True), "konuşamaz": ("konuşmak", False),
}
SORU_EKLERI = ("mı", "mi", "mu", "mü")
TUR_EKLERI = ("tur", "tır", "dur", "dır", "tür", "tir", "dür", "dir")
COGUL_EKLERI = ("lar", "ler")


def tokenize(cumle):
    temiz = "".join(c for c in cumle if c not in NOKTALAMA)
    return temiz.lower().split()


def _ek_at(kelime, ekler):
    for ek in ekler:
        if kelime.endswith(ek) and len(kelime) > len(ek) + 1:
            return kelime[: -len(ek)]
    return kelime


class Niyet:
    def __init__(self, tur, kavram=None, iliski=None, hedef=None, guven=1.0):
        self.tur = tur          # "OGRET" | "SORU" | "BILINMIYOR"
        self.kavram = kavram
        self.iliski = iliski    # "tür" | "yapabilir" | "yapamaz"
        self.hedef = hedef
        self.guven = guven


class SezgiCekirdegi:
    def __init__(self, ag=None):
        self.ag = ag  # Task 6'da MiniAg bağlanır; None ise yalnız kalıp

    def anla(self, cumle):
        t = tokenize(cumle)
        niyet = self._kalip_ayristir(t)
        if self.ag is not None:
            _, guven = self.ag.tahmin(t)
            niyet.guven = guven
        return niyet

    def _kalip_ayristir(self, t):
        # "X nedir"
        if len(t) == 2 and t[1] == "nedir":
            return Niyet("SORU", kavram=t[0], iliski="tür")
        # "X <fiil> mı"
        if len(t) == 3 and t[2] in SORU_EKLERI and t[1] in FIIL_SOZLUGU:
            mastar, _ = FIIL_SOZLUGU[t[1]]
            return Niyet("SORU", kavram=_ek_at(t[0], COGUL_EKLERI),

"""Mini nöral ağ: bag-of-words -> softmax. Tamamen elle: matris yok, kütüphane yok.
llm.docx'teki matematik (softmax, cross-entropy, gradyan) burada canlı."""
import math

SINIFLAR = ["OGRET_TUR", "OGRET_YETENEK", "SORU_YETENEK", "SORU_TANIM"]

_VARLIKLAR = ["kedi", "köpek", "kuş", "balık", "at", "penguen", "serçe", "çocuk"]
_TURLER = ["hayvandır", "kuştur", "canlıdır", "varlıktır"]
_FIILLER = ["uçar", "yüzer", "koşar", "okur", "içer", "konuşur"]
_SORU_EK = ("mı", "mi", "mu", "mü")


def egitim_verisi():
    """Kontrollü dünyanın kalıplarından üretilen (tokenlar, sınıf) örnekleri."""
    ornekler = []
    for v in _VARLIKLAR:
        for t in _TURLER:
            ornekler.append(([v, "bir", t], "OGRET_TUR"))
        for i, f in enumerate(_FIILLER):
            ornekler.append(([v + "lar", f], "OGRET_YETENEK"))
            # soru ekleri dönüşümlü: sınıf dengesi bozulmadan mı/mi/mu/mü öğrenilir
            ornekler.append(([v, f, _SORU_EK[i % 4]], "SORU_YETENEK"))
        ornekler.append(([v, "nedir"], "SORU_TANIM"))
    return ornekler


class MiniAg:
    def __init__(self, siniflar):
        self.siniflar = list(siniflar)
        self.agirlik = {s: {} for s in self.siniflar}   # sınıf -> {kelime: w}
        self.sapma = {s: 0.0 for s in self.siniflar}

    def _logitler(self, tokenlar):
        logit = {}
        for s in self.siniflar:
            w = self.agirlik[s]
            logit[s] = self.sapma[s] + sum(w.get(t, 0.0) for t in tokenlar)
        return logit

    def _softmax(self, logit):
        en_buyuk = max(logit.values())
        us = {s: math.exp(v - en_buyuk) for s, v in logit.items()}
        toplam = sum(us.values())
        return {s: v / toplam for s, v in us.items()}

    def tahmin(self, tokenlar):
        p = self._softmax(self._logitler(tokenlar))
        sinif = max(p, key=p.get)
        return sinif, p[sinif]

    def egit(self, ornekler, tur_sayisi=150, hiz=0.5):
        for _ in range(tur_sayisi):
            for tokenlar, dogru in ornekler:
                p = self._softmax(self._logitler(tokenlar))
                for s in self.siniflar:
                    grad = p[s] - (1.0 if s == dogru else 0.0)  # softmax+CE türevi
                    self.sapma[s] -= hiz * grad
                    for t in tokenlar:
                        self.agirlik[s][t] = self.agirlik[s].get(t, 0.0) - hiz * grad

    @staticmethod
    def varsayilan():
        ag = MiniAg(SINIFLAR)
        ag.egit(egitim_verisi())
        return ag
```

"""Epistemik Kapı: cevap YALNIZCA bellekten üretilir.
Bellekte olmayan şey söylenemez — halüsinasyon filtrelenmez, imkânsızdır."""


class EpistemikKapi:
    def __init__(self, bellek, muhakeme):
        self.bellek = bellek
        self.muhakeme = muhakeme

    def cevapla(self, niyet):
        if niyet.iliski == "tür":
            return self._tanim(niyet.kavram)
        if niyet.iliski == "yapabilir":
            return self._yetenek(niyet.kavram, niyet.hedef)
        return self._bilmiyorum(niyet.kavram)

    def _tanim(self, kavram):
        kenarlar = self.bellek.sorgula(kavram, "tür")
        if not kenarlar:
            return self._bilmiyorum(kavram)
        k = max(kenarlar, key=lambda x: x.guven)
        cevap = f"{kavram} bir {k.hedef}tur (kaynak: {k.kaynak})."
        if k.guven < 0.5:
            return "emin değilim ama " + cevap  # spec §3.3: çekimser cevap
        return cevap

    def _yetenek(self, kavram, eylem):
        sonuc, zincir = self.muhakeme.yapabilir_mi(kavram, eylem)
        if sonuc is None:
            return self._bilmiyorum(kavram)
        on_ek = "evet" if sonuc else "hayır"
        return f"{on_ek}, çünkü {' ve '.join(zincir)}."

    def _bilmiyorum(self, kavram):
        return f"bilmiyorum. {kavram} hakkında bunu bana öğretir misin?"

"""Öğrenme Döngüsü: yeni bilgi -> çelişki kontrolü -> belleğe kalıcı yazım."""
from lmm.bellek import Kenar, DonguHatasi


class OgrenmeDongusu:
    def __init__(self, bellek, muhakeme):
        self.bellek = bellek
        self.muhakeme = muhakeme

    def ogret(self, niyet, kaynak="sen"):
        """Dönüş: (durum, mesaj, kenar). durum: ogrendim|pekisti|celiski|red"""
        aday = Kenar(niyet.kavram, niyet.iliski, niyet.hedef, kaynak=kaynak)
        celiski = self.muhakeme.celiski_kontrol(aday)
        if celiski is not None:
            soru = (f"bir çelişki fark ettim: {celiski}. "
                    f"yine de '{aday.kavram} {aday.hedef} "
                    f"{'yapabilir' if aday.iliski == 'yapabilir' else 'yapamaz'}' "
                    f"olarak öğreneyim mi? (evet/hayır)")
            return "celiski", soru, aday
        mevcut = self.bellek.dogrudan(aday.kavram, aday.iliski, aday.hedef)
        if mevcut is not None:
            self.bellek.yaz(aday)
            return "pekisti", "bunu zaten biliyordum, güvenim arttı.", mevcut
        try:
            self.bellek.yaz(aday)
        except DonguHatasi:
            return "red", "bu tür ilişkisi döngü oluşturur, kabul edemem.", None
        return "ogrendim", f"öğrendim: {aday.kavram} — {aday.iliski} → {aday.hedef}.", aday

    def istisna_onayla(self, kenar):
        """Çelişki kullanıcı onayıyla istisna olarak kalıcılaşır."""
        kenar.istisna = True
        return self.bellek.yaz(kenar)
